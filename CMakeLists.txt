cmake_minimum_required(VERSION 3.22)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(CurveMini VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_executable(curve_tests tests/curve_tests.cpp)
add_executable(motion_tests tests/motion_tests.cpp)
add_test(NAME curves COMMAND curve_tests)
add_test(NAME movements COMMAND motion_tests)
option(CURVES_ONLY "Build original curve and local language tests only" OFF)
if(CURVES_ONLY)
  return()
endif()
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(CurveMini COMPANY_NAME ANKS PLUGIN_MANUFACTURER_CODE Anks PLUGIN_CODE Cpm1
  FORMATS VST3 Standalone PRODUCT_NAME "Curve Mini" IS_SYNTH TRUE
  NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE
  VST3_CATEGORIES Instrument Synth COPY_PLUGIN_AFTER_BUILD FALSE)
juce_generate_juce_header(CurveMini)
target_sources(CurveMini PRIVATE Source/Plugin.cpp Source/Editor.cpp Source/Plugin.h Source/Curve.h Source/MotionLanguage.h)
target_compile_definitions(CurveMini PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_PLUGINHOST_VST3=0)
target_link_libraries(CurveMini PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
if(MSVC)
  target_compile_options(CurveMini PRIVATE /utf-8)
  target_compile_options(motion_tests PRIVATE /utf-8)
endif()
if(BUILD_TESTING)
  add_executable(processor_tests tests/processor_tests.cpp)
  target_include_directories(processor_tests PRIVATE Source $<TARGET_PROPERTY:CurveMini,INCLUDE_DIRECTORIES>)
  target_link_libraries(processor_tests PRIVATE CurveMini)
  add_test(NAME processor COMMAND processor_tests)
endif()
if(BUILD_TESTING AND WIN32)
  juce_add_console_app(vst3_test PRODUCT_NAME "Curve Mini Module Test")
  juce_generate_juce_header(vst3_test)
  target_sources(vst3_test PRIVATE tests/vst3_test.cpp)
  target_compile_definitions(vst3_test PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_MODAL_LOOPS_PERMITTED=1 JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0)
  target_link_libraries(vst3_test PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
  add_dependencies(vst3_test CurveMini_VST3)
  add_test(NAME vst3_module COMMAND vst3_test "$<TARGET_FILE_DIR:CurveMini_VST3>/../..")
  set_tests_properties(vst3_module PROPERTIES TIMEOUT 90)
endif()
